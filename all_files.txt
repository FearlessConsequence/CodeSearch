    inline std::string ParseColorString(const std::string& colorName) {
        if (colorName.empty()) return "";
        std::string ansi = "\033[";
        bool first = true;
        size_t start = 0, end;
        while ((end = colorName.find(' ', start)) != std::string::npos) {
            std::string token = colorName.substr(start, end - start);
            if (token == "bold") ansi += first ? "1" : ";1";
            else if (token == "dim") ansi += first ? "2" : ";2";
            else if (token == "red") ansi += first ? "31" : ";31";
            else if (token == "green") ansi += first ? "32" : ";32";
            else if (token == "yellow") ansi += first ? "33" : ";33";
            else if (token == "blue") ansi += first ? "34" : ";34";
            else if (token == "magenta") ansi += first ? "35" : ";35";
            else if (token == "cyan") ansi += first ? "36" : ";36";
            else if (token == "white") ansi += first ? "37" : ";37";
            first = false;
            start = end + 1;
        }
        std::string token = colorName.substr(start);
        if (token == "bold") ansi += first ? "1" : ";1";
        else if (token == "dim") ansi += first ? "2" : ";2";
        else if (token == "red") ansi += first ? "31" : ";31";
        else if (token == "green") ansi += first ? "32" : ";32";
        else if (token == "yellow") ansi += first ? "33" : ";33";
        else if (token == "blue") ansi += first ? "34" : ";34";
        else if (token == "magenta") ansi += first ? "35" : ";35";
        else if (token == "cyan") ansi += first ? "36" : ";36";
        else if (token == "white") ansi += first ? "37" : ";37";
        return ansi + "m";
    }

    // Применение темы по имени
    inline void ApplyTheme(const std::string& themeName) {
        current_theme.clear();
        auto it = AllThemes.find(themeName);
        if (it != AllThemes.end()) {
            current_theme = it->second;
        } else {
            current_theme = AllThemes["default"]; // Если не нашли, ставим дефолт
        }
    }

    inline const std::string& GetColor(Role role) {
        static const std::string empty = "";
        if (role == Role::Reset) return current_theme[Role::Reset];
        auto it = current_theme.find(role);
        return (it != current_theme.end()) ? it->second : empty;
    }
}

inline std::ostream& operator<<(std::ostream& os, theme::Role role) {
    os << theme::GetColor(role);
    return os;

    static bool MatchesMask(const std::string& fileName, const std::string& mask) {
        auto lower = [](std::string s) { for (auto& c : s) c = std::tolower((unsigned char)c); return s; };
        std::string fn = lower(fileName), m = lower(mask);
        if (m.empty()) return false;
        bool leftStar  = m.front() == '*';
        bool rightStar = m.back()  == '*';
        std::string core = m;
        if (leftStar)  core.erase(core.begin());
        if (rightStar && !core.empty()) core.pop_back();
        if (leftStar && rightStar) return fn.find(core) != std::string::npos;
        if (leftStar)  return fn.size() >= core.size() &&
                            fn.compare(fn.size() - core.size(), core.size(), core) == 0;
        if (rightStar) return fn.compare(0, core.size(), core) == 0;
        return fn == core;
    }

    static bool InVector(const std::vector<std::string>& vec, const std::string& item) {
        for (const auto& v : vec) {
            if (v.size() == item.size()) {
                bool eq = true;
                for (size_t i = 0; i < v.size(); ++i)
                    if (std::tolower((unsigned char)v[i]) != std::tolower((unsigned char)item[i])) { eq = false; break; }
                if (eq) return true;
            }
        }
        return false;
    }

    static std::vector<fs::path> EnumerateFiles(const SearchOptions& opt, size_t& skippedCount) {
        std::vector<fs::path> result;
        try {
            for (auto it = fs::recursive_directory_iterator(opt.Directory, fs::directory_options::skip_permission_denied);
                 it != fs::recursive_directory_iterator(); ++it) {
                const auto& entry = *it;
                if (entry.is_directory()) {
                    std::string name = ToUtf8(entry.path().filename());
                    if (InVector(opt.ExcludeDirs, name)) { it.disable_recursion_pending(); continue; }
                    continue;
                }
                if (!entry.is_regular_file()) continue;
                std::string fileName = ToUtf8(entry.path().filename());
                bool excluded = false;
                for (const auto& mask : opt.ExcludeFiles) if (MatchesMask(fileName, mask)) { excluded = true; break; }
                if (excluded) continue;
                if (!opt.Extensions.empty()) {
                    std::string ext = ToUtf8(entry.path().extension());
                    if (!InVector(opt.Extensions, ext)) continue;
                }
                try {
                    if (opt.MaxFileSizeMb >= 0) {
                        auto bytes = fs::file_size(entry.path());
                        if (bytes > (uintmax_t)opt.MaxFileSizeMb * 1024 * 1024) continue;
                    }
                } catch (...) { skippedCount++; continue; }
                result.push_back(entry.path());
            }
        } catch (...) { skippedCount++; }
        return result;
    }

    static std::string ReadFile(const fs::path& file) {
        std::ifstream in(file, std::ios::binary);
        if (!in) return "";
        std::stringstream buffer;
        buffer << in.rdbuf();
        return buffer.str();
    }

    struct LineTracker {
        const std::string& content;
        size_t pos = 0;
        int line = 1;
        int At(size_t target) {
            if (target < pos) { pos = 0; line = 1; }
            for (size_t i = pos; i < target; ++i) if (content[i] == '\n') line++;
            pos = target;
            return line;
        }
    };

    static std::string TrimmedLineAt(const std::string& content, size_t pos, size_t maxLen = 80) {
        size_t lineStart = content.rfind('\n', pos);
        lineStart = (lineStart == std::string::npos) ? 0 : lineStart + 1;
        size_t lineEnd = content.find('\n', pos);
        if (lineEnd == std::string::npos) lineEnd = content.size();
        std::string text = content.substr(lineStart, lineEnd - lineStart);
        size_t s = text.find_first_not_of(" \t\r");
        size_t e = text.find_last_not_of(" \t\r");
        if (s == std::string::npos) return "";
        text = text.substr(s, e - s + 1);
        if (text.size() > maxLen) {
            size_t rel = (pos >= lineStart + s) ? (pos - lineStart - s) : 0;
            size_t st = (rel > maxLen / 2) ? rel - maxLen / 2 : 0;
            if (st + maxLen > text.size()) st = text.size() - maxLen;
            st = utf8::SafeCut(text, st);
            size_t en = utf8::SafeCut(text, st + maxLen);
            text = "..." + text.substr(st, en - st) + "...";
        }
        return text;
    }

    static void SearchLineByLine(const fs::path& file, const std::regex& regex, std::vector<SearchResult>& results) {
        std::ifstream in(file, std::ios::binary);
        if (!in) return;
        std::string line;
        int lineNum = 0;
        while (std::getline(in, line)) {
            lineNum++;
            if (!line.empty() && line.back() == '\r') line.pop_back();

            // Осознанно: одна строка = одна запись в выводе (см. README, «Ограничения»).
            // Считаем все совпадения, но пушим одно — со счётчиком.
            int count = 0;
            size_t firstPos = std::string::npos;
            for (auto it = std::sregex_iterator(line.begin(), line.end(), regex);
                it != std::sregex_iterator(); ++it) {
                if (count == 0) firstPos = (size_t)it->position();
                ++count;
            }
            if (count == 0) continue;

            size_t start = line.find_first_not_of(" \t");
            size_t end = line.find_last_not_of(" \t");
            std::string trimmed = (start == std::string::npos) ? "" : line.substr(start, end - start + 1);
            if (trimmed.size() > 80) {
                size_t matchRel = (start != std::string::npos && firstPos >= start) ? firstPos - start : 0;
                size_t st = (matchRel > 40) ? matchRel - 40 : 0;
                if (st + 80 > trimmed.size()) st = trimmed.size() - 80;
                st = utf8::SafeCut(trimmed, st);
                size_t en = utf8::SafeCut(trimmed, st + 80);
                trimmed = "..." + trimmed.substr(st, en - st) + "...";
            }
            results.push_back({file, lineNum, trimmed, "", count});
        }
    }
    
    static void SearchWholeFile(const fs::path& file, const std::regex& rx, std::vector<SearchResult>& results) {
        std::string content = ReadFile(file);
        if (content.empty()) return;
        LineTracker lines{content};
        for (auto it = std::sregex_iterator(content.begin(), content.end(), rx); it != std::sregex_iterator(); ++it) {
            size_t pos = it->position();
            int line = lines.At(pos);
            std::string text = TrimmedLineAt(content, pos);
            results.push_back({file, line, text, ""});
        }
    }

    static void SearchStructureXYZ(const fs::path& file, const std::regex& rx, std::vector<SearchResult>& results) {
        std::string content = ReadFile(file);
        if (content.empty()) return;
        LineTracker lines{content};
        for (auto it = std::sregex_iterator(content.begin(), content.end(), rx); it != std::sregex_iterator(); ++it) {
            const auto& m = *it;
            size_t pos = (m.size() > 1 && m[1].matched) ? m.position(1) : m.position();
            int line = lines.At(pos);
            std::string text = TrimmedLineAt(content, pos);
            results.push_back({file, line, text, ""});
        }
    }

    static void SearchXButNotY(const fs::path& file, const std::regex& rxX, const std::regex& rxY, std::vector<SearchResult>& results) {
        std::string content = ReadFile(file);
        if (content.empty()) return;

        std::vector<std::pair<size_t, size_t>> yRanges;
        for (auto it = std::sregex_iterator(content.begin(), content.end(), rxY); it != std::sregex_iterator(); ++it) {
            yRanges.push_back({(size_t)it->position(), (size_t)it->position() + it->length()});
        }

        LineTracker lines{content};
        for (auto it = std::sregex_iterator(content.begin(), content.end(), rxX); it != std::sregex_iterator(); ++it) {
            size_t xs = it->position(), xe = xs + it->length();
            bool intersects = false;
            for (auto& yr : yRanges) if (xs < yr.second && xe > yr.first) { intersects = true; break; }
            if (intersects) continue;
            int line = lines.At(xs);
            results.push_back({file, line, TrimmedLineAt(content, xs), ""});
        }
    }

    static bool IsKeywordBeforeParen(const std::string& w) {
        static const char* kws[] = {
            "if","while","for","switch","catch","using","sizeof","alignof",
            "return","throw","new","delete","typename","static_assert",
            "decltype","co_await","co_return","co_yield","requires","noexcept",
            "constexpr","consteval","constinit","typeid",
            "foreach","lock","nameof","typeof","as","is","when","fixed",
            "checked","unchecked","in","do","else","case","operator"
        };
        for (auto k : kws) if (w == k) return true;
        return false;
    }

    static bool IsWordChar(unsigned char c) {
        return std::isalnum(c) || c == '_';
    }
        
    static bool HasTypeWordBefore(const std::string& content, size_t idStart) {
        size_t p = idStart;
        while (p > 0 && std::isspace((unsigned char)content[p - 1])) --p;
        if (p == 0) return false;

        if (content[p - 1] == '>') {
            // Защита от оператора -> (стрелка), чтобы p->foo() не считалось объявлением
            if (p >= 2 && content[p - 2] == '-') return false;
            return true;  // generic: Foo<int> foo(...)
        }

        if (!IsWordChar((unsigned char)content[p - 1])) return false;
        
        size_t wordEnd = p;
        while (p > 0 && IsWordChar((unsigned char)content[p - 1])) --p;
        std::string word = content.substr(p, wordEnd - p);
        if (word.empty()) return false;
        if (std::isdigit((unsigned char)word[0])) return false;
        if (IsKeywordBeforeParen(word)) return false;
        return true;
    }

    static bool IsCallContextBefore(const std::string& content, size_t idStart) {
        size_t p = idStart;
        while (p > 0 && std::isspace((unsigned char)content[p - 1])) --p;
        if (p == 0) return false;
        char c = content[p - 1];
        static const char* callOps = "=.(,;:?+-*/%&|^~!<";
        bool isOp = false;
        for (const char* q = callOps; *q; ++q) if (*q == c) { isOp = true; break; }
        if (isOp) return true;
        size_t wordEnd = p;
        while (p > 0 && IsWordChar((unsigned char)content[p - 1])) --p;
        if (p < wordEnd) {
            std::string word = content.substr(p, wordEnd - p);
            if (word == "new" || word == "return" || word == "throw" ||
                word == "await" || word == "in" || word == "as" || word == "is" ||
                word == "case" || word == "when" || word == "yield" ||
                word == "co_return" || word == "co_yield" || word == "co_await") {
                return true;
            }
        }
        return false;
    }

    static char CharAfterSpaces(const std::string& content, size_t pos) {
        size_t p = pos + 1;
        while (p < content.size() && std::isspace((unsigned char)content[p])) ++p;
        return (p < content.size()) ? content[p] : 0;
    }

    static int CountArguments(const std::string& inner) {
        int depthParen = 0, depthBrace = 0, depthBracket = 0, depthAngle = 0;
        bool inString = false, inLC = false, inBC = false, hasNonWs = false;
        char sc = 0;
        int commas = 0;

        for (size_t i = 0; i < inner.size(); ++i) {
            char c    = inner[i];
            char next = (i + 1 < inner.size()) ? inner[i + 1] : 0;

            if (inLC) { if (c == '\n') inLC = false; continue; }
            if (inBC) { if (c == '*' && next == '/') { inBC = false; i++; } continue; }
            if (inString) {
                if (c == '\\') { i++; continue; }
                if (c == sc) inString = false;
                continue;
            }
            if (c == '"' || c == '\'') { inString = true; sc = c; hasNonWs = true; continue; }
            if (c == '/' && next == '/') { inLC = true; i++; continue; }
            if (c == '/' && next == '*') { inBC = true; i++; continue; }

            if (!std::isspace((unsigned char)c)) hasNonWs = true;

            if (c == '(') depthParen++;
            else if (c == ')') depthParen--;
            else if (c == '{') depthBrace++;
            else if (c == '}') depthBrace--;
            else if (c == '[') depthBracket++;
            else if (c == ']') depthBracket--;
            else if (c == '<') {
                bool prevIdent = i > 0 && (std::isalnum((unsigned char)inner[i-1]) || inner[i-1] == '_');
                bool nextIdent = next != 0 && (std::isalnum((unsigned char)next) || next == '_');
                if (prevIdent && nextIdent) {
                    int local = 0;
                    for (size_t k = i + 1; k < inner.size(); ++k) {
                        char ch = inner[k];
                        if (ch == '<') local++;
                        else if (ch == '>') {
                            if (local == 0) { depthAngle++; break; }
                            local--;
                        } else if (ch == '(' || ch == ')') break;
                    }
                }
            }
            else if (c == '>' && depthAngle > 0 && (i == 0 || inner[i-1] != '-')) depthAngle--;
            else if (c == ',' && depthParen == 0 && depthBrace == 0
                              && depthBracket == 0 && depthAngle == 0) commas++;
        }
        if (!hasNonWs) return 0;
        return commas + 1;
    }

    static int CountUtf8Chars(const std::string& s) {
        int n = 0;
        for (unsigned char c : s)
            if ((c & 0xC0) != 0x80) n++;
        return n;
    }

    static void SearchLineLength(const fs::path& file,
                                const regex_builder::SearchRequest& req,
                                std::vector<SearchResult>& results) {
        std::ifstream in(file, std::ios::binary);
        if (!in) return;
        std::string line;
        int lineNum = 0;
        while (std::getline(in, line)) {
            lineNum++;
            if (!line.empty() && line.back() == '\r') line.pop_back();
            int len = CountUtf8Chars(line);
            bool ok = false;
            switch (req.NumberCompareValue) {
                case regex_builder::NumberCompare::Greater: ok = (len >  req.MinCount); break;
                case regex_builder::NumberCompare::Equal:   ok = (len == req.MinCount); break;
                case regex_builder::NumberCompare::Less:    ok = (len <  req.MinCount); break;
                case regex_builder::NumberCompare::Between: ok = (len >= req.MinCount && len <= req.MaxCount); break;
            }
            if (!ok) continue;
            size_t start = line.find_first_not_of(" \t");
            size_t end   = line.find_last_not_of(" \t");
            std::string trimmed = (start == std::string::npos) ? "" : line.substr(start, end - start + 1);
            if (trimmed.size() > 80) {
                size_t st = utf8::SafeCut(trimmed, 0);
                size_t en = utf8::SafeCut(trimmed, st + 80);
                trimmed = "..." + trimmed.substr(st, en - st) + "...";
            }
            results.push_back({file, lineNum, trimmed, ""});
        }
    }

    static void SearchComments(const fs::path& file,
                           const regex_builder::SearchRequest& req,
                           std::vector<SearchResult>& results) {
        std::string content = ReadFile(file);
        if (content.empty()) return;

        auto lang = settings::GetLanguage(req.Language);
        std::string prefix = lang.Prefix;

        std::string blockOpen, blockClose;
        if (prefix == "//" || prefix == "--") { blockOpen = "/*"; blockClose = "*/"; }

        struct Comment {
            size_t start, end;          // позиции в файле: от префикса до закрытия
            size_t contentStart, contentEnd;
            bool atLineStart, atLineEnd;
        };
        std::vector<Comment> comments;

        bool inStr = false;
        char sc = 0;
        for (size_t i = 0; i < content.size(); ) {
            char c = content[i];

            // Строки
            if (inStr) {
                if (c == '\\' && i + 1 < content.size()) { i += 2; continue; }
                if (c == sc) inStr = false;
                ++i; continue;
            }
            if (c == '"' || c == '\'') { inStr = true; sc = c; ++i; continue; }

            // Строчный комментарий
            if (!prefix.empty() && c == prefix[0]) {
                bool isLine = true;
                for (size_t k = 1; k < prefix.size(); ++k) {
                    if (i + k >= content.size() || content[i + k] != prefix[k]) { isLine = false; break; }
                }
                if (isLine) {
                    size_t end = content.find('\n', i);
                    if (end == std::string::npos) end = content.size();
                    size_t cs = i + prefix.size();
                    // начало строки: до i только пробелы
                    size_t ls = content.rfind('\n', i);
                    ls = (ls == std::string::npos) ? 0 : ls + 1;
                    bool atStart = true;
                    for (size_t k = ls; k < i; ++k) if (!std::isspace((unsigned char)content[k])) { atStart = false; break; }
                    // конец строки: // всегда до конца
                    comments.push_back({i, end, cs, end, atStart, true});
                    i = end; continue;
                }
            }

            // Блочный комментарий
            if (!blockOpen.empty() && c == blockOpen[0]) {
                bool isBlock = true;
                for (size_t k = 1; k < blockOpen.size(); ++k) {
                    if (i + k >= content.size() || content[i + k] != blockOpen[k]) { isBlock = false; break; }
                }
                if (isBlock) {
                    size_t close = content.find(blockClose, i + blockOpen.size());
                    if (close == std::string::npos) { i = content.size(); break; }
                    size_t cs = i + blockOpen.size();
                    size_t ce = close;
                    // начало строки
                    size_t ls = content.rfind('\n', i);
                    ls = (ls == std::string::npos) ? 0 : ls + 1;
                    bool atStart = true;
                    for (size_t k = ls; k < i; ++k) if (!std::isspace((unsigned char)content[k])) { atStart = false; break; }
                    // конец строки
                    size_t le = content.find('\n', close + blockClose.size());
                    if (le == std::string::npos) le = content.size();
                    bool atEnd = true;
                    for (size_t k = close + blockClose.size(); k < le; ++k)
                        if (!std::isspace((unsigned char)content[k])) { atEnd = false; break; }
                    comments.push_back({i, close + blockClose.size(), cs, ce, atStart, atEnd});
                    i = close + blockClose.size(); continue;
                }
            }
            ++i;
        }

    LineTracker lines{content};
        for (const auto& cm : comments) {
            bool matchMode = false;
            switch (req.Comment) {
                case regex_builder::CommentType::StartLine:     matchMode = cm.atLineStart; break;
                case regex_builder::CommentType::EndLine:       matchMode = cm.atLineEnd;   break;
                case regex_builder::CommentType::CommentedCode: matchMode = true;           break;
            }
            if (!matchMode) continue;

            size_t foundPos = cm.start;
            if (!req.CommentWord.empty()) {
                size_t inBody = content.find(req.CommentWord, cm.contentStart);
                if (inBody == std::string::npos || inBody >= cm.contentEnd) continue;
                foundPos = inBody;
            }

            int line = lines.At(foundPos);
            results.push_back({file, line, TrimmedLineAt(content, foundPos), ""});
        }
    }

    static void SearchArgsLength(const fs::path& file, const regex_builder::SearchRequest& req, std::vector<SearchResult>& results) {
        std::string content = ReadFile(file);
        if (content.empty()) return;

        LineTracker lines{content};
        bool inString = false, inLC = false, inBC = false;
        char sc = 0;

        for (size_t i = 0; i < content.size(); ++i) {
            char c    = content[i];
            char next = (i + 1 < content.size()) ? content[i + 1] : 0;

            if (inLC) { if (c == '\n') inLC = false; continue; }
            if (inBC) { if (c == '*' && next == '/') { inBC = false; i++; } continue; }
            if (inString) {
                if (c == '\\') { i++; continue; }
                if (c == sc) inString = false;
                continue;
            }
            if (c == '"' || c == '\'') { inString = true; sc = c; continue; }
            if (c == '/' && next == '/') { inLC = true; i++; continue; }
            if (c == '/' && next == '*') { inBC = true; i++; continue; }

            if (c != '(') continue;

            size_t jb = i;
            while (jb > 0 && std::isspace((unsigned char)content[jb - 1])) --jb;
            if (jb == 0) continue;

            // Шаблонный вызов: Foo<T>(...) — перед ( стоит >, не являющийся ->
            if (content[jb - 1] == '>' && !(jb >= 2 && content[jb - 2] == '-')) {
                int angleDepth = 0;
                size_t k = jb - 1;
                bool found = false;
                while (k > 0) {
                    char ch = content[k];
                    if (ch == '>') angleDepth++;
                    else if (ch == '<') {
                        angleDepth--;
                        if (angleDepth == 0) { found = true; break; }
                        if (angleDepth < 0) break;
                    }
                    --k;
                }
                if (!found || k == 0) continue;
                size_t beforeLt = k;
                while (beforeLt > 0 && std::isspace((unsigned char)content[beforeLt - 1])) --beforeLt;
                if (beforeLt == 0) continue;
                if (!std::isalnum((unsigned char)content[beforeLt - 1]) && content[beforeLt - 1] != '_') continue;
                jb = beforeLt;
            }

            if (!std::isalnum((unsigned char)content[jb - 1]) && content[jb - 1] != '_') continue;

            size_t idStart = jb;
            while (idStart > 0 && (std::isalnum((unsigned char)content[idStart - 1])
                                   || content[idStart - 1] == '_')) --idStart;
            if (idStart == jb) continue;
            if (std::isdigit((unsigned char)content[idStart])) continue;

            std::string ident = content.substr(idStart, jb - idStart);
            if (IsKeywordBeforeParen(ident)) continue;

            size_t closePos = std::string::npos;
            int depth = 0;
            bool sStr = false, sLC = false, sBC = false;
            char sCh = 0;
            for (size_t j = i; j < content.size(); ++j) {
                char ch  = content[j];
                char nxt = (j + 1 < content.size()) ? content[j + 1] : 0;
                if (sLC) { if (ch == '\n') sLC = false; continue; }
                if (sBC) { if (ch == '*' && nxt == '/') { sBC = false; j++; } continue; }
                if (sStr) {
                    if (ch == '\\') { j++; continue; }
                    if (ch == sCh) sStr = false;
                    continue;
                }
                if (ch == '"' || ch == '\'') { sStr = true; sCh = ch; continue; }
                if (ch == '/' && nxt == '/') { sLC = true; j++; continue; }
                if (ch == '/' && nxt == '*') { sBC = true; j++; continue; }
                if (ch == '(') depth++;
                else if (ch == ')') { depth--; if (depth == 0) { closePos = j; break; } }
            }
            if (closePos == std::string::npos) continue;

            int args = CountArguments(content.substr(i + 1, closePos - i - 1));
            bool ok = false;
            char afterParen = CharAfterSpaces(content, closePos);
            bool hasType = HasTypeWordBefore(content, idStart);
            bool isCallCtx = IsCallContextBefore(content, idStart);

            bool isDecl = false;
            if (hasType && (afterParen == '{' || afterParen == ';' || afterParen == ':')) {
                isDecl = true;
            } else if (afterParen == '{' && !isCallCtx) {
                isDecl = true;
            }
            std::string kind = isDecl ? "decl" : "call";
            switch (req.NumberCompareValue) {
                case regex_builder::NumberCompare::Greater: ok = (args >  req.MinCount); break;
                case regex_builder::NumberCompare::Equal:   ok = (args == req.MinCount); break;
                case regex_builder::NumberCompare::Less:    ok = (args <  req.MinCount); break;
                case regex_builder::NumberCompare::Between: ok = (args >= req.MinCount && args <= req.MaxCount); break;
            }
            if (!ok) continue;

            int line = lines.At(i);
            results.push_back({file, line, TrimmedLineAt(content, i), kind});
        }
    }

    static std::string StripCommentsForCount(const std::string& s) {
        std::string out;
        out.reserve(s.size());
        bool inLC = false, inBC = false, inStr = false;
        char sc = 0;
        for (size_t i = 0; i < s.size(); ++i) {
            char c = s[i];
            char n = (i + 1 < s.size()) ? s[i + 1] : 0;
            if (inLC) { if (c == '\n') { inLC = false; out += '\n'; } continue; }
            if (inBC) { if (c == '*' && n == '/') { inBC = false; i++; } continue; }
            if (inStr) {
                out += c;
                if (c == '\\' && i + 1 < s.size()) { out += s[i + 1]; i++; continue; }
                if (c == sc) inStr = false;
                continue;
            }
            if (c == '"' || c == '\'') { inStr = true; sc = c; out += c; continue; }
            if (c == '/' && n == '/') { inLC = true; i++; continue; }
            if (c == '/' && n == '*') { inBC = true; i++; continue; }
            out += c;
        }
        return out;
    }

    static bool MatchesEmptyMetric(const std::string& inner,
                                const regex_builder::SearchRequest& req) {
        std::string stripped = StripCommentsForCount(inner);
        switch (req.EmptyMetricValue) {
            case regex_builder::EmptyMetric::StrictlyEmpty: {
                for (char c : stripped)
                    if (!std::isspace((unsigned char)c)) return false;
                return true;
            }
            case regex_builder::EmptyMetric::MaxChars: {
                int n = 0;
                for (char c : stripped)
                    if (!std::isspace((unsigned char)c)) n++;
                return n <= req.EmptyMaxCount;
            }
            case regex_builder::EmptyMetric::MaxLines: {
                int n = 0;
                for (char c : stripped)
                    if (c == '\n') n++;
                return n <= req.EmptyMaxCount;
            }
        }
        return false;
    }


    static std::vector<bool> BuildSkipMask(const std::string& s) {
        std::vector<bool> skip(s.size(), false);
        bool inLC = false, inBC = false, inStr = false;
        char sc = 0;
        for (size_t i = 0; i < s.size(); ++i) {
            char c = s[i];
            char n = (i + 1 < s.size()) ? s[i + 1] : 0;
            if (inLC) { skip[i] = true; if (c == '\n') inLC = false; continue; }
            if (inBC) { skip[i] = true; if (c == '*' && n == '/') { inBC = false; skip[i + 1] = true; i++; } continue; }
            if (inStr) {
                skip[i] = true;
                if (c == '\\' && i + 1 < s.size()) { skip[i + 1] = true; i++; continue; }
                if (c == sc) inStr = false;
                continue;
            }
            if (c == '"' || c == '\'') { skip[i] = true; inStr = true; sc = c; continue; }
            if (c == '/' && n == '/') { skip[i] = true; skip[i + 1] = true; inLC = true; i++; continue; }
            if (c == '/' && n == '*') { skip[i] = true; skip[i + 1] = true; inBC = true; i++; continue; }
        }
        return skip;
    }

    static bool HasStatementEndBefore(const std::string& content, size_t start, size_t end) {
        int depth = 0;
        bool inStr = false, inLC = false, inBC = false;
        char sc = 0;
        for (size_t i = start; i < end; ++i) {
            char c = content[i];
            char n = (i + 1 < content.size()) ? content[i + 1] : 0;
            if (inLC) { if (c == '\n') inLC = false; continue; }
            if (inBC) { if (c == '*' && n == '/') { inBC = false; i++; } continue; }
            if (inStr) {
                if (c == '\\' && i + 1 < content.size()) { i++; continue; }
                if (c == sc) inStr = false;
                continue;
            }
            if (c == '"' || c == '\'') { inStr = true; sc = c; continue; }
            if (c == '/' && n == '/') { inLC = true; i++; continue; }
            if (c == '/' && n == '*') { inBC = true; i++; continue; }
            if (c == '(') depth++;
            else if (c == ')') depth--;
            else if (depth == 0 && (c == ';' || c == '}')) return true;
        }
        return false;
    }

    static void SearchEmptyBlocks(const fs::path& file,
                                const regex_builder::SearchRequest& req,
                                std::vector<SearchResult>& results) {
        std::string content = ReadFile(file);
        if (content.empty()) return;
        if (req.X.empty()) return;

        // 1. Собрать все пары { } с учётом вложенности и строк/комментариев.
        std::vector<std::pair<size_t, size_t>> bracePairs;
        std::vector<size_t> openStack;
        {
            bool inLC = false, inBC = false, inStr = false;
            char sc = 0;
            for (size_t i = 0; i < content.size(); ++i) {
                char c = content[i];
                char n = (i + 1 < content.size()) ? content[i + 1] : 0;
                if (inLC) { if (c == '\n') inLC = false; continue; }
                if (inBC) { if (c == '*' && n == '/') { inBC = false; i++; } continue; }
                if (inStr) {
                    if (c == '\\' && i + 1 < content.size()) { i++; continue; }
                    if (c == sc) inStr = false;
                    continue;
                }
                if (c == '"' || c == '\'') { inStr = true; sc = c; continue; }
                if (c == '/' && n == '/') { inLC = true; i++; continue; }
                if (c == '/' && n == '*') { inBC = true; i++; continue; }
                if (c == '{') openStack.push_back(i);
                else if (c == '}') {
                    if (!openStack.empty()) {
                        bracePairs.push_back({openStack.back(), i});
                        openStack.pop_back();
                    }
                }
            }
        }
        if (bracePairs.empty()) return;

        // 2. Regex по X через EscapeXyzWithBounds — понимает %, *, ^, $ и сам ставит \b.
        std::string pattern = regex_builder::EscapeXyzWithBounds(req.X);
        if (pattern.empty()) return;
        std::regex rx;
        try { rx = std::regex(pattern); } catch (...) { return; }

        auto skipMask = BuildSkipMask(content);
        LineTracker lines{content};
        std::set<size_t> seenOpens;

        // 3. Для каждой X — ближайший { после неё, парная }, проверка метрики.
        for (auto it = std::sregex_iterator(content.begin(), content.end(), rx);
            it != std::sregex_iterator(); ++it) {
            size_t xPos = (size_t)it->position();
            if (skipMask[xPos]) continue;
            if (xPos > 0 && content[xPos - 1] == '#') continue;  // #if, #elif...
            size_t xEnd = xPos + it->length();

            if (req.X == "else") {
                size_t p = xEnd;
                while (p < content.size() && std::isspace((unsigned char)content[p])) ++p;
                if (p + 1 < content.size() && content[p] == 'i' && content[p + 1] == 'f'
                    && (p + 2 >= content.size() || !IsWordChar((unsigned char)content[p + 2]))) {
                    continue;
                }
            }

            size_t bestOpen = (size_t)-1;
            for (auto& bp : bracePairs)
                if (bp.first > xEnd && bp.first < bestOpen) bestOpen = bp.first;
            if (bestOpen == (size_t)-1) continue;
            if (HasStatementEndBefore(content, xEnd, bestOpen)) continue;
            if (seenOpens.count(bestOpen)) continue;

            // Привязываем { к ближайшему X: если следующий X идёт до {,
            // то этот блок принадлежит ему, а не текущему (например, if (a) if (b) {}).
            auto nextIt = it;
            ++nextIt;
            while (nextIt != std::sregex_iterator() && skipMask[nextIt->position()]) ++nextIt;
            if (nextIt != std::sregex_iterator() && (size_t)nextIt->position() < bestOpen) continue;

            seenOpens.insert(bestOpen);

            size_t close = (size_t)-1;
            for (auto& bp : bracePairs)
                if (bp.first == bestOpen) { close = bp.second; break; }
            if (close == (size_t)-1) continue;

            std::string inner = content.substr(bestOpen + 1, close - bestOpen - 1);
            if (!MatchesEmptyMetric(inner, req)) continue;

            int line = lines.At((size_t)it->position());
            results.push_back({file, line, TrimmedLineAt(content, (size_t)it->position()), ""});
        }
    }

    static void SearchXButNotYByLine(const fs::path& file,
            const regex_builder::SearchRequest& req,
            std::vector<SearchResult>& results) {
        std::string content = ReadFile(file);
        if (content.empty()) return;

        std::string px = regex_builder::EscapeXyzWithBounds(req.X);
        std::string py = regex_builder::EscapeXyzWithBounds(req.Y);
        if (px.empty() || py.empty()) return;
        std::regex rxX, rxY;
        try {
            rxX = std::regex(px);
            rxY = std::regex(py);
        } catch (...) { return; }

        auto skipMask = BuildSkipMask(content);

        size_t lineStart = 0;
        int lineNum = 0;
        while (lineStart < content.size()) {
            size_t lineEnd = content.find('\n', lineStart);
            if (lineEnd == std::string::npos) lineEnd = content.size();
            lineNum++;

            std::string cleanLine;
            cleanLine.reserve(lineEnd - lineStart);
            for (size_t k = lineStart; k < lineEnd; ++k) {
                cleanLine += skipMask[k] ? ' ' : content[k];
            }
            if (!cleanLine.empty() && cleanLine.back() == '\r') cleanLine.pop_back();

            std::smatch m;
            bool hasY = std::regex_search(cleanLine, m, rxY);
            if (!hasY) {
                // Считаем все вхождения X в строке, чтобы не дублировать записи
                int count = 0;
                for (auto it = std::sregex_iterator(cleanLine.begin(), cleanLine.end(), rxX);
                    it != std::sregex_iterator(); ++it) {
                    count++;
                }
                if (count > 0) {
                    // Пушим одну запись на строку с указанием количества совпадений
                    results.push_back({file, lineNum, TrimmedLineAt(content, lineStart), "", count});
                }
            }

            lineStart = lineEnd + 1;
        }
    }
    std::vector<SearchResult> Search(const regex_builder::SearchRequest& req, const std::string& pattern, const SearchOptions& options, size_t* outFileCount, size_t* outSkippedCount) {
        std::vector<SearchResult> results;
        size_t enumSkipped = 0;
        auto files = EnumerateFiles(options, enumSkipped);
        if (outFileCount) *outFileCount = files.size();
        std::atomic<size_t> skipped{enumSkipped};

        bool isEmptyBlocks = (req.Type == regex_builder::SearchType::Structure &&
                            req.Structure == regex_builder::StructureType::XEmpty);
        bool isStructureXYZ = (req.Type == regex_builder::SearchType::Structure &&
                               req.Structure == regex_builder::StructureType::XAnywhereBeforeYBeforeZ);
        bool isCombinationXOrY = (req.Type == regex_builder::SearchType::Combination &&
                                  req.Combination == regex_builder::CombinationType::XOrY);
        bool isCombinationXButNotY = (req.Type == regex_builder::SearchType::Combination &&
                                      req.Combination == regex_builder::CombinationType::XButNotY);
        bool isArgsLength = (req.Type == regex_builder::SearchType::Quantity &&
                             req.Quantity == regex_builder::QuantityType::ArgsLength);
        bool isLineLength = (req.Type == regex_builder::SearchType::Quantity &&
                     req.Quantity == regex_builder::QuantityType::LineLength);
        bool isComments = (req.Type == regex_builder::SearchType::Comment);
        bool usesPattern = !isArgsLength && !isLineLength && !isComments
                   && !isEmptyBlocks && !isCombinationXButNotY;
        if (usesPattern && pattern.empty()) {
            std::cerr << theme::Role::Error << "Пустое выражение — поиск отменён." << theme::Role::Reset << "\n";
            return results;
        }
        std::regex lineRx, wholeRx, yRx;
        bool needLineRx  = !isArgsLength && !isLineLength && !isComments
                   && !isCombinationXButNotY && !isStructureXYZ
                   && !isEmptyBlocks && !isCombinationXOrY;
        bool needWholeRx = isCombinationXOrY || isStructureXYZ || isCombinationXButNotY;

        try {
            if (isCombinationXButNotY && req.XButNotYModeValue == regex_builder::XButNotYMode::ByPosition) {
                std::string px = regex_builder::EscapeXyzWithBounds(req.X);
                std::string py = regex_builder::EscapeXyzWithBounds(req.Y);
                if (px.empty() || py.empty()) {
                    std::cerr << theme::Role::Error << "Пустое выражение — поиск отменён." << theme::Role::Reset << "\n";
                    return results;
                }
                wholeRx = std::regex(px, std::regex::multiline | std::regex::optimize);
                yRx = std::regex(py, std::regex::multiline | std::regex::optimize);
            } else if (!isCombinationXButNotY) {
                if (needLineRx)  lineRx  = std::regex(pattern, std::regex::optimize);
                if (needWholeRx) wholeRx = std::regex(pattern, std::regex::multiline | std::regex::optimize);
            }
        } catch (const std::regex_error& e) {
            std::cerr << theme::Role::Error << "Ошибка в regex: " << e.what() << theme::Role::Reset << "\n";
            return results;
        }

        std::mutex mtx;
        unsigned int numThreads = std::thread::hardware_concurrency();
        if (numThreads == 0) numThreads = 4;
        if (numThreads > files.size()) numThreads = (unsigned int)std::max<size_t>(1, files.size());
        std::vector<std::thread> threads;
        std::atomic<size_t> index{0};

        auto worker = [&]() {
            std::vector<SearchResult> local;
            while (true) {
                size_t i = index.fetch_add(1);
                if (i >= files.size()) break;
                try {
                    if (isComments)
                        SearchComments(files[i], req, local);
                    else if (isStructureXYZ)
                        SearchStructureXYZ(files[i], wholeRx, local);
                    else if (isEmptyBlocks)
                        SearchEmptyBlocks(files[i], req, local);
                    else if (isCombinationXOrY)
                        SearchWholeFile(files[i], wholeRx, local);
                    else if (isCombinationXButNotY) {
                        if (req.XButNotYModeValue == regex_builder::XButNotYMode::ByLine)
                            SearchXButNotYByLine(files[i], req, local);
                        else
                            SearchXButNotY(files[i], wholeRx, yRx, local);
                    }
                    else if (isArgsLength)
                        SearchArgsLength(files[i], req, local);
                    else if (isLineLength)
                        SearchLineLength(files[i], req, local);
                    else
                        SearchLineByLine(files[i], lineRx, local);
                } catch (...) { skipped++; }
            }
            std::lock_guard<std::mutex> lock(mtx);
            results.insert(results.end(), local.begin(), local.end());
        };

        for (unsigned int t = 0; t < numThreads; ++t) threads.emplace_back(worker);
        for (auto& t : threads) if (t.joinable()) t.join();
        if (outSkippedCount) *outSkippedCount = skipped.load();
        std::sort(results.begin(), results.end(), [](const SearchResult& a, const SearchResult& b) {
            if (a.File != b.File) return a.File < b.File; // Нативное быстрое сравнение путей
            return a.Line < b.Line;
        });
        return results;
    }
}

bool CopyToClipboard(const std::string& text) {
#ifdef _WIN32
    if (!OpenClipboard(nullptr)) return false;
    EmptyClipboard();
    int size = MultiByteToWideChar(CP_UTF8, 0, text.c_str(), -1, nullptr, 0);
    HGLOBAL hMem = GlobalAlloc(GMEM_MOVEABLE, size * sizeof(wchar_t));
    if (!hMem) { CloseClipboard(); return false; }
    wchar_t* mem = (wchar_t*)GlobalLock(hMem);
    MultiByteToWideChar(CP_UTF8, 0, text.c_str(), -1, mem, size);
    GlobalUnlock(hMem);
    SetClipboardData(CF_UNICODETEXT, hMem);
    CloseClipboard();
    return true;
#elif defined(__APPLE__)
    FILE* pipe = popen("pbcopy", "w");
    if (!pipe) return false;
    size_t written = fwrite(text.data(), 1, text.size(), pipe);
    int status = pclose(pipe);
    return written == text.size() && status == 0;
#elif defined(__linux__)
    const char* cmd = nullptr;
    if (getenv("WAYLAND_DISPLAY"))      cmd = "wl-copy";
    else if (getenv("DISPLAY"))         cmd = "xclip -selection clipboard";
    if (!cmd) return false;
    FILE* pipe = popen(cmd, "w");
    if (!pipe) return false;
    size_t written = fwrite(text.data(), 1, text.size(), pipe);
    int status = pclose(pipe);
    return written == text.size() && status == 0;
#else
    (void)text;
    return false;
#endif
}

std::string ReadClipboard() {
#ifdef _WIN32
    if (!OpenClipboard(nullptr)) return "";
    std::string result;
    HANDLE hData = GetClipboardData(CF_UNICODETEXT);
    if (hData) {
        wchar_t* w = (wchar_t*)GlobalLock(hData);
        if (w) {
            int size = WideCharToMultiByte(CP_UTF8, 0, w, -1, nullptr, 0, nullptr, nullptr);
            if (size > 0) {
                std::string buf(size, 0);
                WideCharToMultiByte(CP_UTF8, 0, w, -1, &buf[0], size, nullptr, nullptr);
                buf.resize(size - 1);
                result = buf;
            }
            GlobalUnlock(hData);
        }
    }
    CloseClipboard();
    return result;
#elif defined(__APPLE__)
    FILE* pipe = popen("pbpaste", "r");
    if (!pipe) return "";
    std::string result;
    char buffer[4096];
    size_t n;
    while ((n = fread(buffer, 1, sizeof(buffer), pipe)) > 0) {
        result.append(buffer, n);
    }
    pclose(pipe);
    return result;
#elif defined(__linux__)
    const char* cmd = nullptr;
    if (getenv("WAYLAND_DISPLAY"))      cmd = "wl-paste";
    else if (getenv("DISPLAY"))         cmd = "xclip -selection clipboard -o";
    if (!cmd) return "";
    FILE* pipe = popen(cmd, "r");
    if (!pipe) return "";
    std::string result;
    char buffer[4096];
    size_t n;
    while ((n = fread(buffer, 1, sizeof(buffer), pipe)) > 0) {
        result.append(buffer, n);
    }
    pclose(pipe);
    return result;
#else
    return "";
#endif
}

bool CopyAndRemember(const std::string& text, const std::string& source) {
    (void)source;   // параметр для совместимости, не используется
    std::string oldText = ReadClipboard();
    bool ok = CopyToClipboard(text);
    if (ok && !oldText.empty() && oldText != text) {
        settings::AddClipboardEntry(oldText, "before_copy");
        settings::SaveClipboardHistory();
    }
    return ok;
}

void ShowResults(const std::vector<file_searcher::SearchResult>& results, const file_searcher::SearchOptions& options, int limit) {
    if (results.empty()) {
        std::cout << theme::Role::Dim << "(ничего не найдено)" << theme::Role::Reset << "\n";

    }
    size_t shown = std::min((size_t)limit, results.size());
    for (size_t i = 0; i < shown; ++i) {
        std::string rel;
        try { rel = ToUtf8(fs::relative(results[i].File, options.Directory)); }
        catch (...) { rel = ToUtf8(results[i].File); }
        std::cout << theme::Role::FileName << rel << ":" 
            << theme::Role::LineNumber << results[i].Line << ": ";
        if (!results[i].Kind.empty()) 
            std::cout << theme::Role::Dim << "[" << results[i].Kind << "] " << theme::Role::Reset;
        
        std::cout << theme::Role::Match << results[i].Text << theme::Role::Reset;
        if (results[i].MatchCount > 1)
            std::cout << theme::Role::Dim << " (" << results[i].MatchCount << " "
                      << PluralMatches(results[i].MatchCount) << ")" << theme::Role::Reset;
        std::cout << theme::Role::Reset << "\n"; 
    }
}

static std::string PluralMatches(int n) {
    int mod100 = n % 100;
    int mod10  = n % 10;
    if (mod100 >= 11 && mod100 <= 14) return "совпадений";
    if (mod10 == 1)                   return "совпадение";
    if (mod10 >= 2 && mod10 <= 4)     return "совпадения";
    return "совпадений";
}

void PrintTotals(const std::vector<file_searcher::SearchResult>& results) {
    std::unordered_set<std::string> uniqueFiles;
    uniqueFiles.reserve(results.size());
    for (const auto& r : results) {
        uniqueFiles.insert(ToUtf8(r.File));
    }
    int files = (int)uniqueFiles.size();
    int count = (int)results.size();
    
    std::cout << theme::Role::Question << "Найдено в файлах: " << theme::Role::StatValue << files << theme::Role::Reset << "\n";
    
    theme::Role countColor = theme::Role::StatValue;
    if (count >= 500) countColor = theme::Role::Error;
    else if (count >= 100) countColor = theme::Role::Warning;
    
    std::cout << theme::Role::Question << "Всего найдено: " << countColor << count << theme::Role::Reset << "\n";
}

static std::string DescribeRequest(const regex_builder::SearchRequest& req) {
    switch (req.Type) {
        case regex_builder::SearchType::Word:
            return "Слово: " + req.Word;
        case regex_builder::SearchType::Format: {
            static const char* n[] = { "", "Email", "URL", "IP", "GUID", "Дата" };
                int i = (int)req.Format;
                return std::string("Формат: ") + (i >= 1 && i <= 5 ? n[i] : "?");
            }
        case regex_builder::SearchType::Structure: {
            if (req.Structure == regex_builder::StructureType::XEmpty) {
                std::string metric;
                switch (req.EmptyMetricValue) {
                    case regex_builder::EmptyMetric::StrictlyEmpty:
                        metric = "строго пусто"; break;
                    case regex_builder::EmptyMetric::MaxChars:
                        metric = "≤ " + std::to_string(req.EmptyMaxCount) + " непроб. символов"; break;
                    case regex_builder::EmptyMetric::MaxLines:
                        metric = "≤ " + std::to_string(req.EmptyMaxCount) + " строк"; break;
                }
                return "Структура: X=" + req.X + " (" + metric + ")";
            }
            return "Структура: X=" + req.X + ", Y=" + req.Y +
                (req.Z.empty() ? "" : ", Z=" + req.Z);
        }
        
        case regex_builder::SearchType::Combination: {
            if (req.Combination == regex_builder::CombinationType::XButNotY) {
                std::string mode;
                switch (req.XButNotYModeValue) {
                    case regex_builder::XButNotYMode::ByPosition: mode = "по позиции"; break;
                    case regex_builder::XButNotYMode::ByLine:     mode = "по строке"; break;
                }
                return "Комбинирование: " + req.X + " но не " + req.Y + " (" + mode + ")";
            }
            return "Комбинирование: " + req.X + " / " + req.Y;
        }

        case regex_builder::SearchType::Quantity: {
            static const char* q[] = { "", "цифр в числе", "Длина строки", "Аргументы" };
            int i = (int)req.Quantity;
            std::string cmp;
            switch (req.NumberCompareValue) {
                case regex_builder::NumberCompare::Greater: cmp = "> "  + std::to_string(req.MinCount); break;
                case regex_builder::NumberCompare::Equal:   cmp = "= "  + std::to_string(req.MinCount); break;
                case regex_builder::NumberCompare::Less:    cmp = "< "  + std::to_string(req.MinCount); break;
                case regex_builder::NumberCompare::Between: cmp = std::to_string(req.MinCount) + ".." + std::to_string(req.MaxCount); break;
            }
            return std::string("Количество: ") + (i >= 1 && i <= 3 ? q[i] : "?") + " " + cmp;
        }
        case regex_builder::SearchType::Comment: {
            std::string mode;
            switch (req.Comment) {
                case regex_builder::CommentType::StartLine:     mode = "в начале строки"; break;
                case regex_builder::CommentType::EndLine:       mode = "в конце строки";  break;
                case regex_builder::CommentType::CommentedCode: mode = "все";             break;
            }
            std::string word = req.CommentWord.empty() ? "(все)" : req.CommentWord;
            return "Комментарии (" + mode + ", " + req.Language + "): " + word;
        }
    }
    return "";
}

static bool SaveResultsToFile(const std::vector<file_searcher::SearchResult>& results,
                              const file_searcher::SearchOptions& options,
                              const fs::path& path,
                              const std::string& description) {
    std::ofstream out(path, std::ios::binary);
    if (!out) return false;

    out << "=== CodeSearch ===\n";
    out << "Дата: " << settings::NowString() << "\n";
    out << "Условия: " << description << "\n";
    out << "Найдено: " << results.size() << " совпадений\n";
    out << "------------------------------------------------------------\n";

    for (const auto& r : results) {
        std::string rel;
        try { rel = ToUtf8(fs::relative(r.File, options.Directory)); }
        catch (...) { rel = ToUtf8(r.File); }
        out << rel << ":" << r.Line << ": ";
        if (!r.Kind.empty()) out << "[" << r.Kind << "] ";
        out << r.Text;
        if (r.MatchCount > 1) out << " (" << r.MatchCount << " " << PluralMatches(r.MatchCount) << ")";
        out << "\n";
    }
    return true;
}

        return false;

namespace patterns {

    enum class FormatType {
        Email = 1,
        Url = 2,
        Ip = 3,
        Guid = 4,
        Date = 5
    };

    inline std::string ForFormat(FormatType fmt) {
        switch (fmt) {
            case FormatType::Email:
                return R"([\w.+-]+@[\w-]+\.[\w.-]+)";
            case FormatType::Url:
                return R"(https?://[^\s"']+)";
            case FormatType::Ip:
                return R"((?:^|[^.\d])((?:(?:25[0-5]|2[0-4]\d|1\d\d|[1-9]?\d)\.){3}(?:25[0-5]|2[0-4]\d|1\d\d|[1-9]?\d))(?!\.\d))";
            case FormatType::Guid:
                return R"((?:^|[^\w-])([0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{12})(?![\w-]))";
            case FormatType::Date:
                return R"((?:^|[^\d])((?:\d{4}-(?:0[1-9]|1[0-2])-(?:0[1-9]|[12]\d|3[01])|(?:0[1-9]|[12]\d|3[01])\.(?:0[1-9]|1[0-2])\.\d{4}|\d{4}/(?:0[1-9]|1[0-2])/(?:0[1-9]|[12]\d|3[01])))(?!\d))";
            default:
                return "";
        }
    }
}

    int ReadInt(int minVal, int maxVal) {
        GoBack = false;
        while (true) {
            std::string input;
            std::getline(std::cin, input);

            size_t start = input.find_first_not_of(" \t\r\n");
            size_t end = input.find_last_not_of(" \t\r\n");
            if (start == std::string::npos) {
                std::cout << theme::Role::Question << "> " << theme::Role::Reset;
                continue;
            }
            input = input.substr(start, end - start + 1);

            if (input == "0" && minVal > 0) {
                GoBack = true;
                return minVal;
            }

            try {
                int value = std::stoi(input);
                if (value >= minVal && value <= maxVal)
                    return value;
            }
            catch (...) {}

            std::cout << theme::Role::Question << "> " << theme::Role::Reset;
        }
    }

    std::string ReadLine(const std::string& promptText) {
        std::cout << theme::Role::Question << promptText << theme::Role::Reset;
        GoBack = false; 
        std::string input;
        std::getline(std::cin, input);

        size_t start = input.find_first_not_of(" \t\r\n");
        size_t end = input.find_last_not_of(" \t\r\n");
        if (start == std::string::npos) return "";
        std::string trimmed = input.substr(start, end - start + 1);

        // Если ввели что-то в кавычках — это буквальный текст (например, "0")
        if (trimmed.length() >= 2) {
            if ((trimmed.front() == '"' && trimmed.back() == '"') ||
                (trimmed.front() == '\'' && trimmed.back() == '\'')) {
                return trimmed.substr(1, trimmed.length() - 2);
            }
        }

        if (trimmed == "0") {
            GoBack = true;
            return "";
        }
        return trimmed;
    }

    static std::string FormatSet(const std::vector<std::string>& items) {
        if (items.empty()) return "[пусто]";
        std::string result;
        for (size_t i = 0; i < items.size(); ++i) {
            if (i > 0) result += ", ";
            result += items[i];
        }
        return result;
    }

    static std::vector<std::string> SplitByComma(const std::string& s) {
        std::vector<std::string> result;
        std::stringstream ss(s);
        std::string item;
        while (std::getline(ss, item, ',')) {
            size_t start = item.find_first_not_of(" \t");
            size_t end = item.find_last_not_of(" \t");
            if (start == std::string::npos) continue;
            result.push_back(item.substr(start, end - start + 1));
        }
        return result;
    }

    static bool QuotesBalanced(const std::string& s) {
        bool inQuote = false;
        char q = 0;
        for (char c : s) {
            if (inQuote) { if (c == q) inQuote = false; }
            else if (c == '"' || c == '\'') { inQuote = true; q = c; }
        }
        return !inQuote;
    }

            std::cout << "  " << theme::Role::MenuNum << "1." << theme::Role::Default << " X, но не Y\n";
            std::cout << "  " << theme::Role::MenuNum << "2." << theme::Role::Default << " X или Y или Z\n";
            std::cout << "  " << theme::Role::MenuNum << "0." << theme::Role::Default << " Назад\n" << theme::Role::Question << "> ";
            req.Combination = (regex_builder::CombinationType)ReadInt(1, 2);
            if (GoBack) return false;

            if (req.Combination == regex_builder::CombinationType::XOrY) {
                while (true) {
                    std::cout << theme::Role::Dim
                              << "(% — любые пробелы, * — любые символы до первого совпадения,\n"
                              << " ^ — начало строки, $ — конец строки.\n"
                              << " Если элемент содержит запятую — возьмите его в кавычки \"...\" или '...')"
                              << theme::Role::Reset << "\n";
                    req.X = ReadLine("Элементы через запятую (0 — назад): ");
                    if (GoBack) break;
                    if (!QuotesBalanced(req.X)) {
                        std::cout << theme::Role::Error << "Кавычки не закрыты. Попробуйте снова." << theme::Role::Reset << "\n";
                        continue;
                    }
                    bool hasItem = false;
                    for (char c : req.X) {
                        if (!std::isspace((unsigned char)c) && c != ',' && c != '"' && c != '\'') { hasItem = true; break; }
                    }
                    if (!hasItem) {
                        std::cout << theme::Role::Error << "Нужен хотя бы один элемент." << theme::Role::Reset << "\n";
                        continue;
                    }
                    return true;
                }
                if (GoBack) continue;
            }
            else {
                bool backToTop = false;
                while (true) {
                    std::cout << theme::Role::Question << "Как сравнивать X и Y?\n" << theme::Role::Reset;
                    std::cout << "  " << theme::Role::MenuNum << "1." << theme::Role::Default << " X не пересекается с Y (по позиции)\n";
                    std::cout << "  " << theme::Role::MenuNum << "2." << theme::Role::Default << " В строке с X нет Y (по строке)\n";
                    std::cout << "  " << theme::Role::MenuNum << "0." << theme::Role::Default << " Назад\n" << theme::Role::Question << "> ";
                    req.XButNotYModeValue = (regex_builder::XButNotYMode)ReadInt(1, 2);
                    if (GoBack) { backToTop = true; break; }

                    std::cout << theme::Role::Dim
                              << "(% — любые пробелы, * — любые символы до первого совпадения,\n"
                              << " ^ — начало строки, $ — конец строки)"
                              << theme::Role::Reset << "\n";
                    bool backToSub = false;
                    while (true) {
                        req.X = ReadLine("X (0 — назад): ");
                        if (GoBack) { backToSub = true; break; }
                        req.Y = ReadLine("Y (0 — назад): ");
                        if (GoBack) { backToSub = true; break; }
                        if (req.X.empty()) { std::cout << theme::Role::Error << "X пуст." << theme::Role::Reset << "\n"; continue; }
                        if (req.Y.empty()) { std::cout << theme::Role::Error << "Y пуст." << theme::Role::Reset << "\n"; continue; }
                        req.Exclude = req.Y;
                        return true;
                    }
                    if (backToSub) continue;
                }
                if (backToTop) continue;
            }
        }
    }

    static void ShowHelp() {

            size_t e = input.find_last_not_of(" \t\r\n");
            if (s == std::string::npos) return; // пустой Enter — назад
            input = input.substr(s, e - s + 1);

            std::string lc = input;
            for (auto& c : lc) c = (char)std::tolower((unsigned char)c);

            int n = -1;
            try { n = std::stoi(input); } catch (...) { std::cout << "\n"; continue; }

            if (n == 0) return;
            if (n < 1 || n > 10) { std::cout << "\n"; continue; }

            ShowHelpSection(n);
            std::cout << "\n";
        }

    regex_builder::SearchRequest Run() {
        regex_builder::SearchRequest req;
        GoBack = false;
        std::cout << theme::Role::Title << "=== Умный поиск и генератор regex ===\n" << theme::Role::Reset << "\n";

        bool printMenu = true;
        while (true) {
            if (printMenu) {
                std::cout << theme::Role::Question << "Что искать?\n" << theme::Role::Reset;
                std::cout << "  " << theme::Role::MenuNum << "1." << theme::Role::Default << " Структура\n";
                std::cout << "  " << theme::Role::MenuNum << "2." << theme::Role::Default << " Комбинирование\n";
                std::cout << "  " << theme::Role::MenuNum << "3." << theme::Role::Default << " Количество\n";
                std::cout << "  " << theme::Role::MenuNum << "4." << theme::Role::Default << " Формат (email, URL, IP, GUID, дата)\n";
                std::cout << "  " << theme::Role::MenuNum << "5." << theme::Role::Default << " Комментарии\n";
                std::cout << "  " << theme::Role::MenuNum << "6." << theme::Role::Default << " Слово с границами\n";
                std::cout << "  " << theme::Role::MenuNum << "7." << theme::Role::Default << " Избранное\n";
                std::cout << "  " << theme::Role::MenuNum << "8." << theme::Role::Default << " История поисков\n";
                std::cout << "  " << theme::Role::MenuNum << "9." << theme::Role::Default << " История буфера\n";
                std::cout << "  " << theme::Role::MenuNum << "0." << theme::Role::Default << " Настройки\n";
                std::cout << "  " << theme::Role::MenuNum << "help" << theme::Role::Default << " — справка\n" << theme::Role::Question << "> ";
                printMenu = false;
            } else {
                std::cout << theme::Role::Question << "> " << theme::Role::Reset;
            }

            std::string input;
            std::getline(std::cin, input);
            size_t s = input.find_first_not_of(" \t\r\n");
            size_t e = input.find_last_not_of(" \t\r\n");
            if (s == std::string::npos) continue;
            input = input.substr(s, e - s + 1);

            // help / h / ? / справка — регистронезависимо
            std::string lc = input;
            for (auto& c : lc) c = (char)std::tolower((unsigned char)c);
            if (lc == "help" || lc == "h" || lc == "?" || lc == "справка") {
                ShowHelp();
                printMenu = true;
                continue;
            }

            int choice = -1;
            try { choice = std::stoi(input); } catch (...) { continue; }

            if (choice == 0) {
                EditSettings();
                std::cout << theme::Role::Reset << "\n";
                printMenu = true;
                continue;
            }
            if (choice == 8) {
                WantHistory = true;
                return req;
            }
            if (choice == 7) {
                WantFavorites = true;
                return req;
            }
            if (choice == 9) {
                WantClipboardHistory = true;
                return req;
            }
            if (choice < 1 || choice > 6) continue;

            printMenu = true;

            switch (choice) {
                case 1: req.Type = regex_builder::SearchType::Structure;   break;
                case 2: req.Type = regex_builder::SearchType::Combination; break;
                case 3: req.Type = regex_builder::SearchType::Quantity;    break;
                case 4: req.Type = regex_builder::SearchType::Format;      break;
                case 5: req.Type = regex_builder::SearchType::Comment;     break;
                case 6: req.Type = regex_builder::SearchType::Word;        break;
            }
            std::cout << theme::Role::Reset << "\n";

            bool ok = false;
            switch (req.Type) {
                case regex_builder::SearchType::Word:        ok = AskWord(req); break;
                case regex_builder::SearchType::Format:      ok = AskFormat(req); break;
                case regex_builder::SearchType::Structure:   ok = AskStructure(req); break;
                case regex_builder::SearchType::Combination: ok = AskCombination(req); break;
                case regex_builder::SearchType::Quantity:    ok = AskQuantity(req); break;
                case regex_builder::SearchType::Comment:     ok = AskComment(req); break;
            }

            if (ok) return req;

            GoBack = false;
            std::cout << theme::Role::Reset << "\n";
            printMenu = true;
        }

#include "RegexBuilder.h"
#include <cctype>
#include "Settings.h"
#include <regex>
#include <sstream>
#include <algorithm>

namespace regex_builder {

    const std::string PERCENT_PLACEHOLDER = "\x01" "PERCENT" "\x01";
    const std::string STAR_PLACEHOLDER    = "\x01" "STAR" "\x01";
    const std::string CARET_PLACEHOLDER   = "\x01" "CARET" "\x01";
    const std::string DOLLAR_PLACEHOLDER  = "\x01" "DOLLAR" "\x01";

    std::string Escape(const std::string& s) {
        static const std::string special = R"(\.^$|()[]{}*+?)";
        std::string result;
        for (char c : s) {
            if (special.find(c) != std::string::npos) result += '\\';
            result += c;
        }
        return result;
    }

    std::string EscapeXyz(const std::string& s) {
        std::string temp = s;
        size_t pos;

        while ((pos = temp.find('%')) != std::string::npos) temp.replace(pos, 1, PERCENT_PLACEHOLDER);
        while ((pos = temp.find('*')) != std::string::npos) temp.replace(pos, 1, STAR_PLACEHOLDER);
        while ((pos = temp.find('^')) != std::string::npos) temp.replace(pos, 1, CARET_PLACEHOLDER);
        while ((pos = temp.find('$')) != std::string::npos) temp.replace(pos, 1, DOLLAR_PLACEHOLDER);

        std::string escaped = Escape(temp);

        while ((pos = escaped.find(PERCENT_PLACEHOLDER)) != std::string::npos) escaped.replace(pos, PERCENT_PLACEHOLDER.length(), R"(\s*)");
        while ((pos = escaped.find(STAR_PLACEHOLDER)) != std::string::npos) escaped.replace(pos, STAR_PLACEHOLDER.length(), R"([\s\S]*?)");
        while ((pos = escaped.find(CARET_PLACEHOLDER)) != std::string::npos) escaped.replace(pos, CARET_PLACEHOLDER.length(), R"(^)");
        while ((pos = escaped.find(DOLLAR_PLACEHOLDER)) != std::string::npos) escaped.replace(pos, DOLLAR_PLACEHOLDER.length(), R"($)");
        return escaped;
    }
    
    std::string EscapeXyzWithBounds(const std::string& s) {
        size_t a = s.find_first_not_of(" \t\r\n");
        if (a == std::string::npos) return "";
        size_t b = s.find_last_not_of(" \t\r\n");
        std::string trimmed = s.substr(a, b - a + 1);

        std::string esc = EscapeXyz(trimmed);
        bool leftBound  = std::isalnum((unsigned char)trimmed.front()) || trimmed.front() == '_';
        bool rightBound = std::isalnum((unsigned char)trimmed.back())  || trimmed.back()  == '_';
        if (leftBound)  esc = R"(\b)" + esc;
        if (rightBound) esc = esc + R"(\b)";
        return esc;
    }


    std::string WrapWord(const std::string& s) {
        std::string escaped = Escape(s);
        if (!s.empty() && std::isalpha((unsigned char)s[0])) return R"(\b)" + escaped + R"(\b)";
        return escaped;
    }

    static std::vector<std::string> SplitByCommaWithQuotes(const std::string& s) {
        std::vector<std::string> result;
        std::string current;
        bool inQuote = false;
        char quoteChar = 0;
        for (char c : s) {
            if (inQuote) {
                if (c == quoteChar) inQuote = false;
                else current += c;
            } else {
                if (c == '"' || c == '\'') { inQuote = true; quoteChar = c; }
                else if (c == ',') {
                    size_t a = current.find_first_not_of(" \t");
                    size_t b = current.find_last_not_of(" \t");
                    if (a != std::string::npos) result.push_back(current.substr(a, b - a + 1));
                    current.clear();
                } else current += c;
            }
        }
        size_t a = current.find_first_not_of(" \t");
        size_t b = current.find_last_not_of(" \t");
        if (a != std::string::npos) result.push_back(current.substr(a, b - a + 1));
        return result;
    }

    static std::string BuildWord(const SearchRequest& req) {
        std::string word = Escape(req.Word);
        if (!req.WordWholeOnly) return word;

        bool leftBound  = !req.Word.empty() &&
                        (std::isalnum((unsigned char)req.Word.front()) || req.Word.front() == '_');
        bool rightBound = !req.Word.empty() &&
                        (std::isalnum((unsigned char)req.Word.back())  || req.Word.back()  == '_');

        if (leftBound)  word = R"(\b)" + word;
        if (rightBound) word = word + R"(\b)";
        return word;
    }

    static std::string BuildEmpty(const SearchRequest& req) {
        std::string metric;
        switch (req.EmptyMetricValue) {
            case EmptyMetric::StrictlyEmpty: metric = "строго пусто"; break;
            case EmptyMetric::MaxChars:
                metric = "≤ " + std::to_string(req.EmptyMaxCount) + " непроб. символов"; break;
            case EmptyMetric::MaxLines:
                metric = "≤ " + std::to_string(req.EmptyMaxCount) + " строк"; break;
        }
        return "[empty-block: " + req.X + ", " + metric + "]";
    }

    static std::string BuildStructure(const SearchRequest& req) {
        switch (req.Structure) {
            case StructureType::XEmpty:
                return BuildEmpty(req);
            case StructureType::XAnywhereBeforeYBeforeZ: {
                std::string x = EscapeXyzWithBounds(req.X);
                std::string y = EscapeXyzWithBounds(req.Y);
                std::string z = EscapeXyzWithBounds(req.Z);
                if (x.empty() || y.empty() || z.empty()) return "";
                return x + "[\\s\\S]*?(" + y + ")[\\s\\S]*?" + z;
            }
        }
        return "";
    }

    static std::string BuildCombination(const SearchRequest& req) {
        switch (req.Combination) {
            case CombinationType::XButNotY: {
                std::string mode;
                switch (req.XButNotYModeValue) {
                    case XButNotYMode::ByPosition: mode = "по позиции"; break;
                    case XButNotYMode::ByLine:     mode = "по строке"; break;
                }
                return "[X но не Y (" + mode + "): " + req.X + " / " + req.Y + "]";
            }
            case CombinationType::XOrY: {
                auto items = SplitByCommaWithQuotes(req.X);
                std::string joined;
                for (size_t i = 0; i < items.size(); ++i) {
                    if (i > 0) joined += "|";
                    std::string esc = EscapeXyz(items[i]);
                    bool leftBound = !items[i].empty() && (std::isalnum((unsigned char)items[i][0]) || items[i][0] == '_');
                    bool rightBound = !items[i].empty() && (std::isalnum((unsigned char)items[i].back()) || items[i].back() == '_');
                    if (leftBound) esc = R"(\b)" + esc;
                    if (rightBound) esc = esc + R"(\b)";
                    joined += esc;
                }
                if (joined.empty()) return "";
                return R"((?:)" + joined + R"())";
            }
        }
        return "";
    }

    static std::string BuildNumbers(const SearchRequest& req) {
        switch (req.NumberCompareValue) {
            case NumberCompare::Equal:   return R"(\b\d{)" + std::to_string(req.MinCount) + R"(}\b)";
            case NumberCompare::Greater: return R"(\b\d{)" + std::to_string(req.MinCount + 1) + R"(,}\b)";
            case NumberCompare::Less:
                if (req.MinCount <= 1) return R"((?!))";
                return R"(\b\d{1,)" + std::to_string(req.MinCount - 1) + R"(}\b)";
            case NumberCompare::Between: return R"(\b\d{)" + std::to_string(req.MinCount) + "," + std::to_string(req.MaxCount) + R"(}\b)";
        }
        return "";
    }

    static std::string BuildQuantity(const SearchRequest& req) {
        switch (req.Quantity) {
            case QuantityType::Numbers: return BuildNumbers(req);
            case QuantityType::LineLength:
                switch (req.NumberCompareValue) {
                    case NumberCompare::Greater: return "^.{" + std::to_string(req.MinCount + 1) + ",}$";
                    case NumberCompare::Equal:   return "^.{" + std::to_string(req.MinCount) + "}$";
                    case NumberCompare::Less:
                        if (req.MinCount <= 1) return R"((?!))";
                        return "^.{1," + std::to_string(req.MinCount - 1) + "}$";
                    case NumberCompare::Between: return "^.{" + std::to_string(req.MinCount) + "," + std::to_string(req.MaxCount) + "}$";
                }
                return "";
            case QuantityType::ArgsLength: return "[args-length]";
        }
        return "";
    }

    static std::string BuildTodo(const SearchRequest& req, const std::string& prefix) {
        if (!req.CommentWord.empty()) return prefix + R"(\s*)" + Escape(req.CommentWord) + ".*";
        return prefix + R"(\s*([Tt][Oo][Dd][Oo]|[Ff][Ii][Xx][Mm][Ee]|[Hh][Aa][Cc][Kk]).*)";
    }

    static std::string BuildComment(const SearchRequest& req) {
        std::string mode;
        switch (req.Comment) {
            case CommentType::StartLine:     mode = "в начале строки"; break;
            case CommentType::EndLine:       mode = "в конце строки";  break;
            case CommentType::CommentedCode: mode = "все комментарии"; break;
        }
        std::string word = req.CommentWord.empty() ? "(все)" : req.CommentWord;
        return "[comments: " + mode + ", язык: " + req.Language + ", слово: " + word + "]";
    }

    std::string Build(const SearchRequest& req) {
        switch (req.Type) {
            case SearchType::Word:        return BuildWord(req);
            case SearchType::Format:      return patterns::ForFormat(req.Format);
            case SearchType::Structure:   return BuildStructure(req);
            case SearchType::Combination: return BuildCombination(req);
            case SearchType::Quantity:    return BuildQuantity(req);
            case SearchType::Comment:     return BuildComment(req);
        }
        return "";
    }
}

    inline std::string NowString() {
        std::time_t t = std::time(nullptr);
        std::tm* tm = std::localtime(&t);
        char buf[32];
        std::strftime(buf, sizeof(buf), "%Y-%m-%d %H:%M:%S", tm);
        return std::string(buf);
    }

    inline bool ParseTime(const std::string& s, std::tm& out) {
        std::istringstream ss(s);
        ss >> std::get_time(&out, "%Y-%m-%d %H:%M:%S");
        return !ss.fail();
    }

    inline void CleanClipboardHistory(int hoursKeep = 24) {
        std::time_t now = std::time(nullptr);
        std::vector<ClipboardEntry> kept;
        for (const auto& c : ClipboardHistory) {
            std::tm tm{};
            if (!ParseTime(c.Time, tm)) continue;
            std::time_t t = std::mktime(&tm);
            if (std::difftime(now, t) / 3600.0 <= hoursKeep) kept.push_back(c);
        }
        ClipboardHistory = kept;
    }

    inline void AddClipboardEntry(const std::string& text, const std::string& source) {
        if (text.empty()) return;
        for (auto it = ClipboardHistory.begin(); it != ClipboardHistory.end(); ++it)
            if (it->Text == text) { ClipboardHistory.erase(it); break; }
        ClipboardEntry c;
        c.Time = NowString();
        c.Text = text;
        c.Source = source;
        ClipboardHistory.insert(ClipboardHistory.begin(), c);
        while ((int)ClipboardHistory.size() > MaxClipboardEntries) ClipboardHistory.pop_back();
    }

    auto _a = (actual); auto _e = (expected); \
    if (!(_a == _e)) { \
        std::cerr << "  FAIL " << __FILE__ << ":" << __LINE__ \
                  << "  expected [" << _e << "] got [" << _a << "]\n"; \
        ++g_failed; \
    } else { ++g_passed; } \
} while (0)

static bool RxHit(const std::string& pattern, const std::string& text) {
    try { return std::regex_search(text, std::regex(pattern)); }
    catch (...) { return false; }
}

static void Test_BuildQuantity_Numbers() {
    Section("BuildQuantity (Numbers)");
    auto make = [](regex_builder::NumberCompare c, int mn, int mx) {
        regex_builder::SearchRequest r;
        r.Type = regex_builder::SearchType::Quantity;
        r.Quantity = regex_builder::QuantityType::Numbers;
        r.NumberCompareValue = c;
        r.MinCount = mn; r.MaxCount = mx;
        return regex_builder::Build(r);
    };
    using NC = regex_builder::NumberCompare;

    // Equal
    std::string eq3 = make(NC::Equal, 3, 0);
    CHECK_TRUE (RxHit(eq3, "999"));
    CHECK_FALSE(RxHit(eq3, "99"));
    CHECK_FALSE(RxHit(eq3, "1000"));
    CHECK_FALSE(RxHit(eq3, "12a"));

    // Greater
    std::string gt3 = make(NC::Greater, 3, 0);

struct TempDir {
    fs::path path;
    TempDir() {
        path = fs::temp_directory_path() / ("cs_test_" + std::to_string(std::rand()));
        fs::create_directories(path);
    }
    ~TempDir() { std::error_code ec; fs::remove_all(path, ec); }
    void Write(const std::string& name, const std::string& content) const {
        std::ofstream out(path / name, std::ios::binary);
        out << content;
    }
    void WritePath(const std::string& rel, const std::string& content) const {
        fs::path p = path / rel;
        if (p.has_parent_path() && p.parent_path() != path) {
            std::error_code ec;
            fs::create_directories(p.parent_path(), ec);
        }
        std::ofstream out(p, std::ios::binary);
        out << content;
    }
};

static file_searcher::SearchOptions MakeOptions(const fs::path& dir) {
    file_searcher::SearchOptions o;
    o.Directory = dir;
    o.Extensions = {".cs"};
    o.ExcludeDirs = {};
    o.ExcludeFiles = {};
    o.MaxFileSizeMb = -1;
    o.ShowFirstN = 1000;
    return o;
}

#pragma once
#include <filesystem>
#include <string>

namespace utf8 {
    inline std::string ToUtf8(const std::filesystem::path& p) {
        auto u8 = p.u8string();
        std::string result;
        result.reserve(u8.size());
        for (auto c : u8) result += static_cast<char>(c);
        return result;
    }

    inline size_t SafeCut(const std::string& s, size_t pos) {
        while (pos > 0 && pos < s.size() && ((unsigned char)s[pos] & 0xC0) == 0x80) pos--;
        return pos;
    }

    inline std::filesystem::path FromUtf8(const std::string& s) {
#if defined(__cpp_lib_char8_t)
        return std::filesystem::path(std::u8string(s.begin(), s.end()));
#else
        return std::filesystem::u8path(s);
#endif
    }
}

struct Color {
    int code;
    int r, g, b;
    std::string hex;
};

std::string ToHex(int r, int g, int b) {
    char buf[10];
    snprintf(buf, sizeof(buf), "#%02x%02x%02x", r, g, b);
    return std::string(buf);
}

// Функция, которая применяет алгоритм "Ближайший сосед" к любому списку
void ApplyTSP(std::vector<Color>& list) {
    if (list.empty()) return;
    std::vector<Color> sorted;
    std::vector<bool> visited(list.size(), false);
    
    // Находим самый темный цвет, чтобы начать с него (или черный)
    int current = 0; 
    double min_l = 999;
    for (size_t i = 0; i < list.size(); ++i) {
        double l = 0.299 * list[i].r + 0.587 * list[i].g + 0.114 * list[i].b;
        if (l < min_l) {
            min_l = l;
            current = i;
        }
    }

    sorted.push_back(list[current]);
    visited[current] = true;

    for (size_t step = 1; step < list.size(); ++step) {
        int next_color = -1;
        double min_dist = 999999.0;

        for (size_t i = 0; i < list.size(); ++i) {
            if (visited[i]) continue;
            
            double dr = list[current].r - list[i].r;
            double dg = list[current].g - list[i].g;
            double db = list[current].b - list[i].b;
            double dist = 2.0 * dr * dr + 4.0 * dg * dg + 3.0 * db * db; 

            if (dist < min_dist) {
                min_dist = dist;
                next_color = i;
            }
        }

        if (next_color != -1) {
            current = next_color;
            visited[current] = true;
            sorted.push_back(list[current]);
        }
    }
    list = sorted; // Заменяем исходный массив на отсортированный
}
